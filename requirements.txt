#   Linux / Ubuntu:
#       sudo apt-get install -y ncbi-blast+ default-jre
#       # KMC: download a release binary from
#       #   https://github.com/refresh-bio/KMC/releases  (place under bin/bin/)
#       curl -s https://get.nextflow.io | bash && sudo mv nextflow /usr/local/bin/
#
#   Any platform with conda (recommended, one command):
#       conda install -c bioconda -c conda-forge kmc blast nextflow
#
#   Windows:
#       Use WSL2 (Ubuntu) and follow the Linux instructions, or run the whole
#       project inside the conda environment.
